        echo 错误: 模型文件不存在！

echo ========================================
echo 模型管理
echo ========================================

    goto model_management

if errorlevel 1 (
    echo 重命名失败！
) else (
    echo 重命名成功！
)

:delete_model

call :list_models
echo ----------------------------------------
set /p del_choice=请选择要删除的模型 (1-%count%)：

set "del_name=!model%del_choice%!"
set "del_path=%model_path%\%del_name%"

echo.
echo 警告: 即将删除模型: %del_name%

dir "%del_path%" /-c | findstr "%del_name%"

set /p confirm=确认删除？(输入 YES 确认):

if /i "%confirm%"=="YES" (
    del "%del_path%"

        echo 删除失败！
    ) else (
        echo 删除成功！
    )
) else (
    echo 操作已取消
)

:check_space
cls
echo ========================================
echo 磁盘空间信息
echo ========================================
echo 模型文件夹: %model_path%
echo.
wmic logicaldisk where "caption='%model_path:~0,2%'" get deviceid,size,freespace /format:table
echo.
echo 模型文件夹大小:
powershell -command "Get-ChildItem -Path '%model_path%' -Recurse | Measure-Object -Property Length -Sum | ForEach-Object { '{0:N2} GB' -f ($_.Sum / 1GB) }" 2>nul
if errorlevel 1 echo 无法计算大小（需要PowerShell支持）

goto model_management

if errorlevel 1 (
    echo 备份失败！
) else (
    echo 备份成功！已保存到 %backup_dest%
)

:: ==========================================
:: 子函数

:list_models
set count=0
echo 可用的模型列表:
echo ----------------------------------------
for /f "delims=" %%i in ('dir /b /a-d "%model_path%\*" 2^>nul') do (
    set /a count+=1
    set "model!count!=%%i"
    echo !count!. %%i
)
exit /b

:validate_input
set "input=%~1"
set "max=%2"
echo %input%|findstr /r "^[0-9]*$" >nul
if errorlevel 1 (
    echo 无效输入，请输入数字！
    timeout /t 2 >nul
    exit /b 1
)
if %input% lss 1 (
    echo 输入不能小于1！
    timeout /t 2 >nul
    exit /b 1
)
if %input% gtr %max% (
    echo 输入超出范围！
    timeout /t 2 >nul
    exit /b 1
)
exit /b 0

:run_interactive
set "timestamp=%date:~0,4%%date:~5,2%%date:~8,2%_%time:~0,2%%time:~3,2%%time:~6,2%"
set "timestamp=%timestamp: =0%"
set "log_file=%log_dir%\conversation_%timestamp%.log"

echo ========================================
echo 启动交互式对话模式
echo ========================================
echo 模型: %selected_model%
echo.
echo 使用说明:
echo - 直接输入问题，按回车发送
echo - 按 Ctrl+C 退出对话
echo ========================================
echo.

:: 直接运行模型，不使用 tee（Windows不支持）
if defined cli_preset (
    echo 使用预设参数: %preset_name%
    "%cli_exe%" -m "%full_model_path%" --color auto %cli_preset%
) else (

    echo 下载失败！
    echo 可能原因: 模型ID错误或需要授权

    echo 如果是需要授权的模型，请先登录:

    echo ========================================

echo ========================================
echo 下载GGUF格式模型
echo ========================================
echo.

echo 推荐使用 TheBloke 等作者提供的GGUF版本

:: 使用Python脚本获取GGUF文件列表并下载
python -c "
import subprocess
import sys
import os

model_id = r'%gguf_id%'
target_dir = r'%model_path%\%gguf_id:\=%'
target_dir = target_dir.replace('/', '\\')

# 设置镜像
os.environ['HF_ENDPOINT'] = 'https://hf-mirror.com'

try:
    from huggingface_hub import list_repo_files, hf_hub_download
    
    print('正在获取文件列表...')
    files = list_repo_files(model_id)
